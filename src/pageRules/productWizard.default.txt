async function entry(context) {
    /**
     * context is an object that contains the following information
     * {
     *     currentQuestion: number,
     *     user: {
     *         name       : string,
     *         username   : string,
     *         role       : ?{
     *             id: string,
     *         },
     *         permissions: Array<string>,
     *     },
     *     answers: Array<{
     *          question: string,
     *          elements: Array<{
     *              field: string,
     *              value: string,
     *          }>,
     *          metaData?: mixed,
     *     }>
     * }
     *
     * context.currentQuestion is a number starting at zero (first question)
     */

    const pluralize = (name) => {
        if(name.slice(-1) === "s") {
            return name + "'";
        }

        return name + "'s";
    };

    let product = {
        barcode     : '',
        name        : '',
        category    : null,
        tags        : [],
        caseQuantity: '',
        prices      : [],
    };

    for(let i = 0, l = context.answers.length; i < l; i++) {
        if(typeof context.answers[i].metaData !== 'undefined') {
            if(context.answers[i].metaData === "barcode") {
                let variable = false;
                try {
                    variable = window.ShopfrontAPI.getVariable('hasSetBarcode');
                } catch {}

                if(variable !== true && variable !== context.answers[i].elements[0].value) {
                    window.ShopfrontAPI.setField('barcodes', [{
                        quantity: 1,
                        code    : context.answers[i].elements[0].value,
                    }]);

                    window.ShopfrontAPI.storeVariable('hasSetBarcode', context.answers[i].elements[0].value);
                }

                product.barcode = context.answers[i].elements[0].value;
            } else if(context.answers[i].metaData === "prices") {
                product.prices = [{
                    quantity: context.answers[i].elements[0].value,
                    price   : context.answers[i].elements[1].value,
                }];
            } else {
                product[context.answers[i].metaData] = context.answers[i].elements[0].value;
            }
        }
    }

    if(product.barcode && context.currentQuestion === 1) {
        // Warn when the barcode already belongs to a product in this store.
        const matching = await window.ShopfrontAPI.findByField("products", "barcode", product.barcode);

        if(matching.data.length !== 0) {
            window.ShopfrontAPI.toast("info", `That barcode is already being used by ${matching.data[0].name}`);
        }
    }

    const questions = [{
        question     : 'What is the barcode for the product?',
        questionsLeft: 5,
        title        : "Create Product",
        elements     : [{
            type    : 'text',
            field   : 'barcode',
            default : product.barcode,
            required: false,
        }],
        metaData: "barcode",
    }, {
        question     : 'What is the name of this product?',
        questionsLeft: 4,
        title        : "Create Product",
        elements     : [{
            type    : 'text',
            field   : 'name',
            default : product.name,
            required: true,
        }],
        metaData: 'name',
    }, {
        question     : `What category does ${product.name} belong to?`,
        questionsLeft: 3,
        title        : "Create Product",
        elements     : [{
            type      : 'select',
            field     : 'category',
            default   : product.category ? product.category.value : null,
            required  : false,
            dataSource: 'categories',
            __label   : product.category ? product.category.label : null,
        }],
        metaData: 'category',
    }, {
        question     : `What tags does ${product.name} have?`,
        questionsLeft: 2,
        title        : "Create Product",
        elements     : [{
            type      : 'select',
            field     : 'tags',
            default   : product.tags,
            required  : false,
            dataSource: 'tags',
            multiple  : true,
        }],
        metaData: 'tags',
    }, {
        question     : `How many ${pluralize(product.name)} come in a case?`,
        questionsLeft: 1,
        title        : "Create Product",
        elements     : [{
            type    : 'number',
            field   : 'caseQuantity',
            default : product.caseQuantity,
            step    : 1,
            min     : 1,
            required: true,
        }],
        metaData: 'caseQuantity',
    }, {
        question     : `What prices does ${product.name} have?`,
        questionsLeft: 0,
        title        : "Create Product",
        metaData     : 'prices',
        elements     : [{
            type       : 'number',
            field      : 'prices.0.quantity',
            placeholder: 'Quantity',
            default    : product.prices.length >= 1 ? product.prices[0].quantity : 1,
            step       : 1,
            min        : 1,
            required   : true,
        }, {
            type       : 'currency',
            field      : 'prices.0.price',
            placeholder: 'Price',
            default    : product.prices.length >= 1 ? product.prices[0].price : '',
            required   : true,
        }, {
            type         : "custom",
            field        : "footnote",
            customElement: {
                type      : 'p',
                textNode  : 'You can customise the prices further on the next page.',
                attributes: {
                    style: {
                        color: '#ffffff',
                    },
                },
            }
        }],
    }];

    /**
     * It is expected that the function returns an object with the following signature:
     * {
     *     question: string,
     *     title?  : string,
     *     elements: Array<{
     *         type        : string,
     *         field       : string,
     *         placeholder?: string,
     *         default?    : string,
     *         creatable?  : boolean,
     *         required?   : boolean,
     *         step?       : number,
     *         min?        : number,
     *         max?        : number,
     *         options?    : Array<{
     *             label: string,
     *             value: string,
     *         }> | string,
     *     }>,
     *     questionsLeft: number,
     *     metaData?    : mixed,
     * }
     */
    return questions[context.currentQuestion];
}
