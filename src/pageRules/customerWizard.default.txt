async function entry(context) {
    /**
     * context is an object that contains the following information
     * {
     *     user: {
     *         name       : string,
     *         username   : string,
     *         role       : ?{
     *             id: string,
     *         },
     *         permissions: Array<string>,
     *     },
     *     answers: Array<{
     *          question: string,
     *          elements: Array<{
     *              field: string,
     *              value: string,
     *          }>,
     *          metaData?: mixed,
     *     }>
     * }
     */

    const pluralize = (name) => {
        if(name.slice(-1) === "s") {
            return name + "'";
        }

        return name + "'s";
    };

    if(context.currentQuestion === 0) {
        window.__customerWizard = {
            code: "",
        };
    }

    let name = "the customer";
    if(context.answers.length > 0) {
        if(context.answers[0].elements[0].value) {
            name = context.answers[0].elements[0].value;

            if(context.answers[0].elements[1].value) {
                name += ' ';
            }
        }

        if(context.answers[0].elements[1].value) {
            name += context.answers[0].elements[1].value;
        }
    }

    const customer = {
        firstName    : context.answers[0] ? context.answers[0].elements[0].value : "",
        lastName     : context.answers[0] ? context.answers[0].elements[1].value : "",
        customerGroup: context.answers[1] ? context.answers[1].elements[0].value : "",
        code         : context.answers[2] ? context.answers[2].elements[0].value : "",
        email        : context.answers[3] ? context.answers[3].elements[0].value : "",
    };

    const questions = [{
        question     : `What is the customer's name?`,
        title        : "Create Customer",
        questionsLeft: 3,
        elements     : [{
            type       : 'text',
            field      : 'firstName',
            placeholder: 'First Name (required)',
            required   : true,
            default    : customer.firstName
        }, {
            type       : 'text',
            field      : 'lastName',
            placeholder: 'Last Name',
            default    : customer.lastName,
        }],
    }, {
        question     : `Which group does ${name} belong to?`,
        title        : "Create Customer",
        questionsLeft: 2,
        elements     : [{
            type             : 'select',
            field            : 'customerGroup',
            dataSource       : 'customerGroups',
            minimumSearchSize: 0,
            default          : customer.customerGroup ? customer.customerGroup.value : null,
            __label          : customer.customerGroup ? customer.customerGroup.label : null,
        }],
    }, {
        question     : `What code would you like to allocate to ${name}?`,
        title        : "Create Customer",
        questionsLeft: 1,
        elements     : [{
            type       : 'text',
            field      : 'code',
            placeholder: 'Code',
            default    : customer.code,
        }],
    }, {
        question     : `What is ${pluralize(name)} email?`,
        title        : "Create Customer",
        questionsLeft: 0,
        elements     : [{
            type       : 'text',
            field      : 'emails.0',
            placeholder: 'Email',
            default    : customer.email,
        }],
    }];

    if(customer.code && customer.code !== window.__customerWizard.code) {
        window.__customerWizard.code = customer.code;
        const matching = await window.ShopfrontAPI.findByField("customers", "code", customer.code);

        if(matching.data.length !== 0) {
            const matchingCustomers = [];
            for(let i = 0, l = matching.data.length; i < l; i++) {
                matchingCustomers.push(`${matching.data[i].firstName} ${matching.data[i].lastName}`);
            }

            window.ShopfrontAPI.toast("info", `That customer code is already being used by ${matchingCustomers.join(", ")}`);
        }
    }

    /**
     * It is expected that the function returns an object with the following signature:
     * {
     *     question: string,
     *     title?  : string,
     *     elements: Array<{
     *         type        : string,
     *         field       : string,
     *         placeholder?: string,
     *         default?    : string,
     *         creatable?  : boolean,
     *         required?   : boolean,
     *         options?    : Array<{
     *             label: string,
     *             value: string,
     *         }> | string,
     *     }>,
     *     questionsLeft: number,
     *     metaData?    : mixed,
     * }
     */
    return questions[context.currentQuestion];
}
